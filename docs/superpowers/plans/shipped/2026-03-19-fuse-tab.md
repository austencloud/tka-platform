# Fuse Tab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A new "Fuse" tab in the Create module where users select two sequences side by side with live animation previews, then fuse them into a combined sequence with an assembly animation and celebration effect.

**Architecture:** Five-phase state machine (browse -> left-selected -> both-selected -> fusing -> result) with independent AnimationPlaybackControllers per panel, synced by shared BPM. Assembly animation uses FLIP technique with Web Animations API. State follows Factory + Context pattern.

**Tech Stack:** Svelte 5, ITI DI, AnimatorCanvas, Web Animations API, BrowseGrid

**Spec:** `docs/superpowers/specs/2026-03-19-fuse-tab.md`

**Depends on:** `2026-03-19-compositional-data-model.md` plan (specifically Task 13: SequenceFuser)

```typescript
// src/lib/features/fuse/state/fuse-state.svelte.ts
import type { SequenceData } from "$lib/shared/foundation/domain/models/SequenceData";
import type { IAnimationPlaybackController } from "$lib/features/compose/services/contracts/IAnimationPlaybackController";
import type { ISequenceFuser } from "$lib/features/fuse/services/contracts/ISequenceFuser";

export type FusePhase = "browse" | "left-selected" | "both-selected" | "fusing" | "result";

export interface FuseDeps {
  sequenceFuser: ISequenceFuser;
}

export function createFuseState(deps: FuseDeps) {
  let phase = $state<FusePhase>("browse");
  let leftSequence = $state<SequenceData | null>(null);
  let rightSequence = $state<SequenceData | null>(null);
  let fusedSequence = $state<SequenceData | null>(null);
  let matchLengths = $state(true);
  let bpm = $state(60);

  // Playback controllers and animation state (created when sequences are selected)
  let leftController = $state<IAnimationPlaybackController | null>(null);
  let rightController = $state<IAnimationPlaybackController | null>(null);
  // AnimationPanelState instances are needed by AnimationPlaybackController.initialize()
  // Create these when a sequence is selected, dispose when deselected

  const canFuse = $derived(
    phase === "both-selected" &&
    leftSequence !== null &&
    rightSequence !== null
  );

  function selectLeft(seq: SequenceData) {
    leftSequence = seq;
    phase = rightSequence ? "both-selected" : "left-selected";
  }

  function selectRight(seq: SequenceData) {
    rightSequence = seq;
    phase = leftSequence ? "both-selected" : "left-selected";
  }

  function deselectLeft() {
    leftSequence = null;
    phase = rightSequence ? "left-selected" : "browse";
  }

  function deselectRight() {
    rightSequence = null;
    phase = leftSequence ? "left-selected" : "browse";
  }

  function startFuse() {
    if (!leftSequence || !rightSequence) return;
    phase = "fusing";

    // SequenceFuser.fuse() takes HandPathData | SoloPropData for each side.
    // Extract the blue solo prop from the left sequence and red from the right.
    // These fields exist because SequenceHydrator.ensureComposition() populates them
    // at save time (data model plan Tasks 1-5).
    const blue = leftSequence.blueSoloProp;
    const red = rightSequence.redSoloProp;

    if (blue && red) {
      fusedSequence = deps.sequenceFuser.fuse(blue, red);
    }
  }

  function completeFuse() {
    phase = "result";
  }

  function reset() {
    phase = "browse";
    leftSequence = null;
    rightSequence = null;
    fusedSequence = null;
  }

  function setBpm(newBpm: number) {
    bpm = newBpm;
  }

  function setMatchLengths(match: boolean) {
    matchLengths = match;
  }

  return {
    get phase() { return phase; },
    get leftSequence() { return leftSequence; },
    get rightSequence() { return rightSequence; },
    get fusedSequence() { return fusedSequence; },
    get matchLengths() { return matchLengths; },
    get bpm() { return bpm; },
    get canFuse() { return canFuse; },
    selectLeft,
    selectRight,
    deselectLeft,
    deselectRight,
    startFuse,
    completeFuse,
    reset,
    setBpm,
    setMatchLengths,
  };
}

export type FuseState = ReturnType<typeof createFuseState>;
```

```typescript
// src/lib/features/fuse/services/implementations/FuseAssemblyAnimator.ts
import type { IFuseAssemblyAnimator } from "../contracts/IFuseAssemblyAnimator";

export class FuseAssemblyAnimator implements IFuseAssemblyAnimator {
  private readonly DURATION = 600;
  private readonly EASING = "cubic-bezier(0.16, 1, 0.3, 1)";

  async animate(
    leftEl: HTMLElement,
    rightEl: HTMLElement,
    targetEl: HTMLElement
  ): Promise<void> {
    const leftRect = leftEl.getBoundingClientRect();
    const rightRect = rightEl.getBoundingClientRect();
    const targetRect = targetEl.getBoundingClientRect();

    const leftTransform = this.calcFlipTransform(leftRect, targetRect);
    const rightTransform = this.calcFlipTransform(rightRect, targetRect);

    const opts: KeyframeAnimationOptions = {
      duration: this.DURATION,
      easing: this.EASING,
      fill: "both" as FillMode,
    };

    const leftAnim = leftEl.animate([
      { transform: "translate(0, 0) scale(1)", opacity: 1 },
      { transform: `translate(${leftTransform.tx}px, ${leftTransform.ty}px) scale(${leftTransform.sx}, ${leftTransform.sy})`, opacity: 0 },
    ], opts);

    const rightAnim = rightEl.animate([
      { transform: "translate(0, 0) scale(1)", opacity: 1 },
      { transform: `translate(${rightTransform.tx}px, ${rightTransform.ty}px) scale(${rightTransform.sx}, ${rightTransform.sy})`, opacity: 0 },
    ], opts);

    await Promise.all([leftAnim.finished, rightAnim.finished]);

    // Clean up
    leftAnim.cancel();
    rightAnim.cancel();
  }

  private calcFlipTransform(from: DOMRect, to: DOMRect) {
    const fromCenterX = from.left + from.width / 2;
    const fromCenterY = from.top + from.height / 2;
    const toCenterX = to.left + to.width / 2;
    const toCenterY = to.top + to.height / 2;

    return {
      tx: toCenterX - fromCenterX,
      ty: toCenterY - fromCenterY,
      sx: to.width / from.width,
      sy: to.height / from.height,
    };
  }
}
```

## Task 10: Celebration Effect (CelebrationOverlay)

**Files:**
- Create: `src/lib/features/fuse/components/CelebrationOverlay.svelte`

- [ ] **Step 1: Write the particle celebration component**

Canvas-based particle system. Orange/amber particles radiating from center, 800ms duration.

```svelte
<!-- src/lib/features/fuse/components/CelebrationOverlay.svelte -->
<script lang="ts">
  import { onMount } from "svelte";

  interface Props {
    centerX: number;
    centerY: number;
    onComplete: () => void;
  }

  let { centerX, centerY, onComplete }: Props = $props();
  let canvas: HTMLCanvasElement;

  interface Particle {
    x: number;
    y: number;
    vx: number;
    vy: number;
    life: number;
    maxLife: number;
    size: number;
    color: string;
  }

  const COLORS = ["#fb923c", "#f97316", "#ea580c", "#fbbf24", "#f59e0b"];
  const PARTICLE_COUNT = 50;
  const DURATION = 800;

  onMount(() => {
    if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) {
      setTimeout(onComplete, 100);
      return;
    }

    const ctx = canvas.getContext("2d");
    if (!ctx) return;

    canvas.width = canvas.offsetWidth * devicePixelRatio;
    canvas.height = canvas.offsetHeight * devicePixelRatio;
    ctx.scale(devicePixelRatio, devicePixelRatio);

    const particles: Particle[] = Array.from({ length: PARTICLE_COUNT }, () => {
      const angle = Math.random() * Math.PI * 2;
      const speed = 2 + Math.random() * 6;
      return {
        x: centerX,
        y: centerY,
        vx: Math.cos(angle) * speed,
        vy: Math.sin(angle) * speed - 2,
        life: 1,
        maxLife: 0.6 + Math.random() * 0.4,
        size: 3 + Math.random() * 4,
        color: COLORS[Math.floor(Math.random() * COLORS.length)],
      };
    });

    const start = performance.now();
    let frameId: number;

    function frame(now: number) {
      const elapsed = now - start;
      const progress = Math.min(elapsed / DURATION, 1);

      ctx!.clearRect(0, 0, canvas.offsetWidth, canvas.offsetHeight);

      for (const p of particles) {
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.15; // gravity
        p.life = Math.max(0, 1 - progress / p.maxLife);

        if (p.life <= 0) continue;

        ctx!.globalAlpha = p.life;
        ctx!.fillStyle = p.color;
        ctx!.beginPath();
        ctx!.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
        ctx!.fill();
      }

      ctx!.globalAlpha = 1;

      if (progress < 1) {
        frameId = requestAnimationFrame(frame);
      } else {
        onComplete();
      }
    }

    frameId = requestAnimationFrame(frame);

    return () => cancelAnimationFrame(frameId);
  });
</script>

<canvas
  bind:this={canvas}
  class="celebration-canvas"
  aria-hidden="true"
></canvas>

<style>

    position: absolute;
    inset: 0;
    width: 100%;
    height: 100%;
    pointer-events: none;
    z-index: 10;
  }
</style>
```
