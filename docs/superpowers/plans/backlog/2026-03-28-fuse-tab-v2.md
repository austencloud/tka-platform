# Fuse Tab v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rewrite the Fuse tab as a single component that transitions between assembled/disassembled states using FLIP animation, eliminating the hard DOM swap between FuseLayout and FuseResultView.

**Architecture:** One component (`FuseLayout`) manages the entire lifecycle with three always-mounted `AnimatorCanvas` instances (hero, blue-only, red-only). All three canvases are always in the DOM (hidden via `visibility: hidden`, never `{#if}`). State machine: `disassembled ↔ reassembling ↔ assembled ↔ disassembling`. FLIP animations drive the visual transitions. No component swap, no canvas remounting.

**Tech Stack:** Svelte 5 (runes), Web Animations API (FLIP technique), AnimatorCanvas, ChoreoCard, ITI DI

**Spec:** `docs/superpowers/specs/2026-03-28-fuse-tab-v2-design.md`

---

## Critical Design Decisions

### All Three Canvases Always Mounted

The hero, blue-only, and red-only `AnimatorCanvas` instances are ALWAYS in the DOM. They are never conditionally rendered with `{#if}`. Visibility is controlled with `visibility: hidden` + `position: absolute` so `getBoundingClientRect()` always returns measurable rects for FLIP calculations. Canvases that are hidden pass `isPlaying={false}` to avoid GPU work.

### FusePanel Bypassed for Canvas Rendering

`FusePanel` currently wraps `FuseAnimationPreview` which creates its own `AnimatorCanvas`. In v2, the three `AnimatorCanvas` instances are owned by `FuseLayout` directly (they must persist across state changes). `FusePanel` is replaced by inlining `FuseSequenceBrowser` (beat grid) and the shuffle button directly in `FuseLayout`. `FusePanel.svelte` and `FuseAnimationPreview.svelte` become dead code and are deleted.

### Prop State Management

Each `AnimatorCanvas` needs reactive `blueProp` / `redProp` state. Currently `FuseAnimationPreview` handles this through `ISequenceMotionLoader` + `calculateStateForBeat()`. In v2, `FuseLayout` creates a shared `createAnimationPanelState()` for the hero canvas. The split canvases reuse the panel controllers from `FuseSequenceBrowser` (which already drive the per-side previews). When the hero canvas needs prop states, it uses the fused sequence + the same `createAnimationPanelState()` pattern from the deleted `FuseResultView`.

```typescript
$effect(() => {
  const seq = fuseState.fusedSequence;
  if (!seq || !heroController || !motionLoader || !isAssembled) return;

  untrack(async () => {
    if (heroAnimState.isPlaying) heroController!.togglePlayback();
    heroAnimState.reset();

    const fullSeq = await motionLoader!.ensureMotionData(seq);
    if (!fullSeq) return;

    heroAnimState.setShouldLoop(true);
    const ok = heroController!.initialize(fullSeq, heroAnimState);
    if (!ok) return;

    const speed = fuseState.bpm / 60;
    heroController!.setSpeed(speed);
    setTimeout(() => heroController?.togglePlayback(), 300);
  });
});
```

```typescript
const FLIP_DURATION = 600;
const FLIP_EASING = "cubic-bezier(0.16, 1, 0.3, 1)";

function calcFlipTransform(
  sourceRect: DOMRect,
  targetRect: DOMRect
) {
  const sCX = sourceRect.left + sourceRect.width / 2;
  const sCY = sourceRect.top + sourceRect.height / 2;
  const tCX = targetRect.left + targetRect.width / 2;
  const tCY = targetRect.top + targetRect.height / 2;
  return {
    tx: tCX - sCX, ty: tCY - sCY,
    sx: targetRect.width / sourceRect.width,
    sy: targetRect.height / sourceRect.height,
  };
}

function cancelAllAnimations(elements: (HTMLElement | undefined)[]) {
  for (const el of elements) {
    if (!el) continue;
    for (const anim of el.getAnimations()) anim.cancel();
  }
}
```

- [ ] **Step 2: Add reassemble animation**

```typescript
function animateReassemble() {
  if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) {
    fuseState.completeReassemble();
    return;
  }
  if (!heroEl || !splitLeftEl || !splitRightEl) {
    fuseState.completeReassemble();
    return;
  }

  // FLIP measurements (all elements are in DOM, some visibility:hidden)
  const heroRect = heroEl.getBoundingClientRect();
  const leftRect = splitLeftEl.getBoundingClientRect();
  const rightRect = splitRightEl.getBoundingClientRect();

  const leftFlip = calcFlipTransform(leftRect, heroRect);
  const rightFlip = calcFlipTransform(rightRect, heroRect);

  const opts: KeyframeAnimationOptions = {
    duration: FLIP_DURATION, easing: FLIP_EASING, fill: "both",
  };

  // Fade out browse chrome
  for (const el of [browseLeftEl, browseRightEl]) {
    el?.animate([{ opacity: 1 }, { opacity: 0 }], {
      duration: 200, easing: "ease-out", fill: "both",
    });
  }

  // FLIP split canvases toward hero
  splitLeftEl.animate([
    { transform: "translate(0,0) scale(1)" },
    { transform: `translate(${leftFlip.tx}px,${leftFlip.ty}px) scale(${leftFlip.sx},${leftFlip.sy})` },
  ], opts);
  splitLeftEl.animate([{ opacity: 1 }, { opacity: 0 }], {
    duration: FLIP_DURATION * 0.25, delay: FLIP_DURATION * 0.75, easing: "ease-in", fill: "both",
  });

  const mainAnim = splitRightEl.animate([
    { transform: "translate(0,0) scale(1)" },
    { transform: `translate(${rightFlip.tx}px,${rightFlip.ty}px) scale(${rightFlip.sx},${rightFlip.sy})` },
  ], opts);
  splitRightEl.animate([{ opacity: 1 }, { opacity: 0 }], {
    duration: FLIP_DURATION * 0.25, delay: FLIP_DURATION * 0.75, easing: "ease-in", fill: "both",
  });

  mainAnim.finished.then(() => {
    cancelAllAnimations([heroEl, splitLeftEl, splitRightEl, browseLeftEl, browseRightEl]);
    fuseState.completeReassemble();

    // Card entrance
    if (cardAreaEl) {
      cardAreaEl.animate([
        { opacity: 0, transform: "translateX(30px)" },
        { opacity: 1, transform: "translateX(0)" },
      ], { duration: 200, easing: FLIP_EASING, fill: "both" });
    }
  });
}
```

```typescript
function animateDisassemble() {
  if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) {
    fuseState.completeDisassemble();
    return;
  }
  if (!heroEl || !splitLeftEl || !splitRightEl) {
    fuseState.completeDisassemble();
    return;
  }

  // Card exit
  if (cardAreaEl) {
    cardAreaEl.animate([
      { opacity: 1, transform: "translateX(0)" },
      { opacity: 0, transform: "translateX(30px)" },
    ], { duration: 200, easing: "ease-in", fill: "both" });
  }

  setTimeout(() => {
    const heroRect = heroEl.getBoundingClientRect();
    const leftRect = splitLeftEl.getBoundingClientRect();
    const rightRect = splitRightEl.getBoundingClientRect();

    const leftFlip = calcFlipTransform(leftRect, heroRect);
    const rightFlip = calcFlipTransform(rightRect, heroRect);

    const opts: KeyframeAnimationOptions = {
      duration: FLIP_DURATION, easing: FLIP_EASING, fill: "both",
    };

    // FLIP splits from hero position to natural positions
    splitLeftEl.animate([
      { transform: `translate(${leftFlip.tx}px,${leftFlip.ty}px) scale(${leftFlip.sx},${leftFlip.sy})`, opacity: 0 },
      { transform: "translate(0,0) scale(1)", opacity: 1 },
    ], opts);
    splitLeftEl.animate([{ opacity: 0 }, { opacity: 1 }], {
      duration: FLIP_DURATION * 0.25, easing: "ease-out", fill: "both",
    });

    const mainAnim = splitRightEl.animate([
      { transform: `translate(${rightFlip.tx}px,${rightFlip.ty}px) scale(${rightFlip.sx},${rightFlip.sy})`, opacity: 0 },
      { transform: "translate(0,0) scale(1)", opacity: 1 },
    ], opts);
    splitRightEl.animate([{ opacity: 0 }, { opacity: 1 }], {
      duration: FLIP_DURATION * 0.25, easing: "ease-out", fill: "both",
    });

    mainAnim.finished.then(() => {
      cancelAllAnimations([heroEl, splitLeftEl, splitRightEl, browseLeftEl, browseRightEl]);

      // Fade in browse chrome
      for (const el of [browseLeftEl, browseRightEl]) {
        el?.animate([{ opacity: 0 }, { opacity: 1 }], {
          duration: 200, easing: "ease-out", fill: "both",
        });
      }

      fuseState.completeDisassemble();
    });
  }, 200); // Wait for card exit
}
```
